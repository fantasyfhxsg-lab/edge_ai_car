cmake_minimum_required(VERSION 3.16)
project(edge_ai_car LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -Wall -Wextra")
if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -march=armv8-a+simd")
endif()
find_package(OpenCV REQUIRED)
find_package(Threads REQUIRED)
include_directories(${CMAKE_SOURCE_DIR}/include ${OpenCV_INCLUDE_DIRS})
add_executable(aeb_system src/main.cpp src/v4l2_capture.cpp src/preprocess.cpp
    src/inference_engine.cpp src/logger.cpp)
target_link_libraries(aeb_system ${OpenCV_LIBS} Threads::Threads)
install(TARGETS aeb_system DESTINATION bin)
